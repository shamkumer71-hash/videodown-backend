const express = require("express"); const cors = require("cors"); const app = express(); app.use(cors()); app.use(express.json()); /* ========================= HOME / API STATUS ========================= */ app.get("/", (req, res) => { res.json({ status: "online", service: "VideoDown API", version: "1.0.0" }); }); /* ========================= PLATFORM DETECTOR ========================= */ function detectPlatform(url) { try { const host = new URL(url) .hostname .toLowerCase() .replace(/^www\./, ""); if ( host === "youtube.com" || host === "m.youtube.com" || host === "youtu.be" ) { return "YouTube"; } if ( host === "facebook.com" || host === "m.facebook.com" || host === "fb.watch" ) { return "Facebook"; } if ( host === "kuaishou.com" || host === "kwai.com" ) { return "Kuaishou"; } return null; } catch (error) { return null; } } /* ========================= ANALYZE URL ========================= */ app.post("/api/analyze", (req, res) => { const { url } = req.body; if (!url) { return res.status(400).json({ success: false, message: "Video URL is required" }); } const platform = detectPlatform(url); if (!platform) { return res.status(400).json({ success: false, message: "Unsupported or invalid video URL" }); } res.json({ success: true, platform: platform, message: "Platform detected successfully." }); }); /* ========================= VIDEO INFO ========================= */ app.post("/api/video-info", (req, res) => { const { url } = req.body; if (!url) { return res.status(400).json({ success: false, message: "Video URL is required" }); } const platform = detectPlatform(url); if (!platform) { return res.status(400).json({ success: false, message: "Unsupported or invalid video URL" }); } /* এখানে ভবিষ্যতে বৈধ/public content processing বা official API integration যুক্ত করা হবে। */ res.json({ success: true, platform: platform, status: "ready", message: "Video source detected. Authorized/public content processing can be connected here.", availableQualities: [ "360p", "480p", "720p", "1080p" ] }); }); /* ========================= DOWNLOAD STATUS ========================= */ app.post("/api/download", (req, res) => { const { url, quality } = req.body; if (!url) { return res.status(400).json({ success: false, message: "Video URL is required" }); } const platform = detectPlatform(url); if (!platform) { return res.status(400).json({ success: false, message: "Unsupported or invalid video URL" }); } const allowedQualities = [ "360p", "480p", "720p", "1080p" ]; if (!allowedQualities.includes(quality)) { return res.status(400).json({ success: false, message: "Invalid video quality" }); } res.json({ success: true, platform: platform, quality: quality, status: "pending", message: "Download processing endpoint is ready for an authorized/public content downloader implementation." }); }); /* ========================= 404 ERROR ========================= */ app.use((req, res) => { res.status(404).json({ success: false, message: "API endpoint not found" }); }); /* ========================= START SERVER ========================= */ const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log( `VideoDown API running on port ${PORT}` ); });
